cmake_minimum_required(VERSION 3.20)
project(log_hunter_native LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows SDK and MSVC are required")
endif()
set_source_files_properties(vss_requester.c PROPERTIES LANGUAGE CXX)
add_library(log_hunter_native SHARED shared_read.c eventlog_export.c registry_save.c prefetch_decompress.c vss_requester.c)
target_compile_definitions(log_hunter_native PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0601)
target_compile_options(log_hunter_native PRIVATE /W4)
target_link_libraries(log_hunter_native PRIVATE vssapi ole32 oleaut32 advapi32 wevtapi)
set_target_properties(log_hunter_native PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
